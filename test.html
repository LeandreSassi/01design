// Ik heb uit de vorige tutorial alles gekopieerd om makkelijk verder te kunnen
let img;
let modulo;
let epilepsie = 0;
function preload() {
  img = loadImage("Informatica p5js.jpg");
  frameRate(30);
}
function setup() {
  createCanvas(img.width, img.height);
  pixelDensity(1);
}

function draw() {
  image(img, 0, 0, img.width, img.height);
  var modulo = 16+47*Math.sin(epilepsie)**2;
  //console.log(modulo, epilepsie)
  loadPixels();
  for (var y = 0; y < height; y++) {
    for (var x = 0; x < width; x++) {
      var index = (x + y * width) * 4;
      var r = pixels[index + 0];
      var g = pixels[index + 1];
      var b = pixels[index + 2];
      var a = pixels[index + 3];
      
      if (r % modulo < 10) {
        pixels[index + 0] = 255;
      }
      if (g % modulo < 10) {
        pixels[index + 1] = 255;
      }
      if (b % modulo < 10) {
        pixels[index + 2] = 255;
      }
    }
  }
  updatePixels();
  epilepsie += 0.05;
}
